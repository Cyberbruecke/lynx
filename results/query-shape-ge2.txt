// Wrapper: crawler/out/js/ is archived as js.tar.zst, so query_shape.js's onDisk()
// check fails for every sha. Substitute an equivalent "script is in the corpus" test:
// presence of out/tools/lynx/<sha>.json (132,646 entries == the paper's script count).
const fs = require("fs"), path = require("path");
const TOOLS_LYNX = "/workspace/crawler/out/tools/lynx";
const real = fs.existsSync.bind(fs);
const JSDIR = "/workspace/crawler/out/js/";
fs.existsSync = (p) => {
  const s = String(p);
  if (s.startsWith(JSDIR) && s.endsWith(".js"))
    return real(path.join(TOOLS_LYNX, path.basename(s, ".js") + ".json"));
  return real(p);
};
process.argv[2] = process.env.NDOM || "100000";
require("/workspace/crawler/query_shape.js");
